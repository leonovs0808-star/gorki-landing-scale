    treeContainer.style.alignItems = 'stretch';
    treeContainer.style.justifyContent = 'center';
    treeContainer.style.position = 'relative';
    treeContainer.style.height = '100%';
    treeContainer.style.minHeight = containerHeight;
    // Make tree image fill the full height on ALL screens
    var treeImgEl = treeContainer.querySelector('img');
    if (treeImgEl) {
      treeImgEl.style.width = '100%';
      treeImgEl.style.height = '100%';
      treeImgEl.style.objectFit = 'cover';
      treeImgEl.style.objectPosition = 'center center';
      if (isMobile) {
        treeImgEl.style.objectPosition = 'right center';
      }
    }

    // Position strategies absolutely
    strategies.forEach(function(child, idx) {
      if (idx >= 3) return;
      child.style.position = 'absolute';
      child.style.top = positions[idx].top;
      child.style.left = positions[idx].left;
      child.style.right = positions[idx].right;
      child.style.opacity = '1';
      child.style.transform = 'none';
      child.style.display = 'flex';
      child.style.flexDirection = 'column';
      child.style.maxWidth = isMobile ? '70%' : '';
      child.style.overflow = 'hidden';
    });

    // Mobile font adjustments
    if (isMobile) {
      flexParent.querySelectorAll('h3').forEach(function(h) {
        h.style.fontSize = isSmallMobile ? '0.85rem' : '1rem';
        h.style.marginBottom = '0.25rem';
        h.style.lineHeight = '1.3';
        h.style.fontWeight = '700';
      });
      flexParent.querySelectorAll('[class*="flex-wrap"] span, [class*="flex-wrap"] div').forEach(function(tag) {
        if (tag.tagName === 'SPAN' || (tag.className && tag.className.indexOf('rounded') !== -1)) {
          tag.style.fontSize = isSmallMobile ? '0.45rem' : '0.5rem';
          tag.style.padding = '0.1rem 0.25rem';
          tag.style.margin = '0.05rem';
          tag.style.whiteSpace = 'nowrap';
        }
      });
      // Green lines — thinner on mobile
      flexParent.querySelectorAll('[class*="bg-primary"][class*="rounded-full"]').forEach(function(dot) {
        var w = parseInt(window.getComputedStyle(dot).width);
        if (w > 10) { dot.style.width = '8px'; dot.style.height = '8px'; }
      });
      // Green lines (the horizontal bars)
      flexParent.querySelectorAll('[class*="bg-primary"][class*="h-"]').forEach(function(line) {
        var h = parseInt(window.getComputedStyle(line).height);
        if (h <= 3 && parseInt(window.getComputedStyle(line).width) > 50) {
          line.style.height = '1px';
        }
      });
    }
    return;

    // Desktop/tablet
    {
      // Desktop/tablet: absolute positioning
      flexParent.style.flexDirection = '';
      flexParent.style.alignItems = 'stretch';
      flexParent.style.minHeight = '500px';
      flexParent.style.padding = '';
      // Tree container reset
      var treeContainer = treeImg.parentElement;
      treeContainer.style.width = '45%';
      treeContainer.style.maxWidth = '';
      treeContainer.style.margin = '';
      treeContainer.style.order = '';
      var positions = [
        { top: '28%', left: '30%', right: '4%' },
        { top: '58%', left: '20%', right: '4%' },
        { top: '74%', left: '28%', right: '4%' }
      ];
      strategies.forEach(function(child, idx) {
        if (idx < 3) {
          child.style.position = 'absolute';
          child.style.top = positions[idx].top;
          child.style.left = positions[idx].left;
          child.style.right = positions[idx].right;
          child.style.width = '';
          child.style.opacity = '1';
          child.style.marginBottom = '';
          child.style.textAlign = '';
        }
      });
    }
  }
  fixTreeLayout();
  window.addEventListener('resize', fixTreeLayout);

  /* ===== FIX CAROUSEL DOTS SPACING ===== */
  var firstDot = document.querySelector('[data-carousel-dot]');
  if (firstDot && firstDot.parentElement) {
    firstDot.parentElement.style.marginTop = '1.5rem';
    firstDot.parentElement.style.paddingBottom = '1rem';
  }

  /* ===== MOBILE BURGER MENU ===== */
  var burgerBtn = document.querySelector('button.lg\\:hidden');
  if (burgerBtn) {
    var navLinksContainer = document.querySelector('nav .hidden.lg\\:flex');
    // Create mobile menu
    var mobileMenu = document.createElement('div');
    mobileMenu.id = 'mobile-menu';
    mobileMenu.style.cssText = 'display:none;position:fixed;top:60px;left:0;right:0;bottom:0;background:hsl(160,55%,5%);z-index:49;padding:2rem;';
    if (navLinksContainer) {
      var links = navLinksContainer.querySelectorAll('a');
      links.forEach(function(link) {
        var a = document.createElement('a');
        a.href = link.href;
        a.textContent = link.textContent;
        a.style.cssText = 'display:block;color:hsl(155,35%,94%);font-size:1.125rem;padding:0.75rem 0;border-bottom:1px solid hsl(160,35%,12%);text-decoration:none;';
        a.addEventListener('click', function() {
          mobileMenu.style.display = 'none';
          burgerBtn.setAttribute('data-state', 'closed');
        });
        mobileMenu.appendChild(a);
      });
    }
    // Add CTA button to mobile menu
    var ctaLink = document.createElement('a');
    ctaLink.href = '#cta-form';
    ctaLink.textContent = 'Записаться';
    ctaLink.style.cssText = 'display:inline-block;margin-top:1.5rem;padding:0.75rem 2rem;background:hsl(160,45%,40%);color:white;border-radius:9999px;text-decoration:none;font-weight:600;text-align:center;';
    ctaLink.addEventListener('click', function() {
      mobileMenu.style.display = 'none';
      burgerBtn.setAttribute('data-state', 'closed');
    });
    mobileMenu.appendChild(ctaLink);
    document.body.appendChild(mobileMenu);

    burgerBtn.addEventListener('click', function() {
      var isOpen = mobileMenu.style.display !== 'none';
      mobileMenu.style.display = isOpen ? 'none' : 'block';
      this.setAttribute('data-state', isOpen ? 'closed' : 'open');
    });
  }

  /* ===== SPEAKER HOVER TOOLTIPS ===== */
  var speakerData = {
    'Владимир Седов': { role: 'Основатель Askona Life Group и города Доброград', desc: 'Вырастил «Аскону» с 1,2 млрд до 100+ млрд выручки, 11 000 сотрудников, 19 стран.' },
    'Евгения Курчатова': { role: 'Со-основатель школы бизнеса «Горки»', desc: 'Сквозной педагог программы. Приземляет кейсы педагогов на реальность каждого студента.' },
    'Александр Барсуков': { role: 'Предприниматель, экс-CEO сети «Магнит»', desc: 'От товароведа в Ростове до CEO «Магнита»: 9+ бизнес-моделей, 300 000 сотрудников.' },
    'Михаил Воробьев': { role: 'Сооснователь «ЭР-Телеком» (бренд Дом.ru)', desc: 'Построил компанию с нуля до 100+ млрд выручки.' },
    'Игорь Хведук': { role: 'Основатель PERMEGI', desc: 'Лидер категории товаров для дома на маркетплейсах.' },
    'Дмитрий Тарасевич': { role: 'Со-основатель пивоваренной компании «ТАРКОС»', desc: '15+ лет стабильного роста из Воронежа, без московских ресурсов.' },
    'Виктор Семёнов': { role: 'Председатель НС группы компаний «Белая Дача»', desc: 'Создал рынок готовых салатов в России. Экс-министр сельского хозяйства РФ.' }
  };

  var tooltip = document.createElement('div');
  tooltip.style.cssText = 'position:fixed;z-index:50;width:14rem;padding:0.75rem;border-radius:0.75rem;border:1px solid hsl(160,45%,40%,0.3);background:hsl(160,35%,12%);box-shadow:0 10px 25px rgba(0,0,0,0.3);pointer-events:none;opacity:0;transition:opacity 0.2s;';
  document.body.appendChild(tooltip);

  document.querySelectorAll('.group.cursor-pointer').forEach(function(card) {
    var img = card.querySelector('img[alt]');
    if (!img || !speakerData[img.alt]) return;
    var data = speakerData[img.alt];

    card.addEventListener('mouseenter', function(e) {
      tooltip.innerHTML = '<h4 style="font-weight:700;font-size:0.875rem;color:hsl(155,35%,94%)">' + img.alt + '</h4>'
        + '<p style="color:hsl(160,45%,40%);font-size:0.75rem;font-weight:500;margin-top:2px">' + data.role + '</p>'
        + '<p style="color:hsl(160,20%,60%);font-size:0.75rem;margin-top:0.5rem;line-height:1.5">' + data.desc + '</p>';
      tooltip.style.opacity = '1';
    });
    card.addEventListener('mousemove', function(e) {
      tooltip.style.left = (e.clientX + 12) + 'px';
      tooltip.style.top = (e.clientY + 12) + 'px';
    });
    card.addEventListener('mouseleave', function() {
      tooltip.style.opacity = '0';
    });
    // Touch support
    card.addEventListener('click', function(e) {
      if (tooltip.style.opacity === '1') {
        tooltip.style.opacity = '0';
      } else {
        var rect = card.getBoundingClientRect();
        tooltip.innerHTML = '<h4 style="font-weight:700;font-size:0.875rem;color:hsl(155,35%,94%)">' + img.alt + '</h4>'
          + '<p style="color:hsl(160,45%,40%);font-size:0.75rem;font-weight:500;margin-top:2px">' + data.role + '</p>'
          + '<p style="color:hsl(160,20%,60%);font-size:0.75rem;margin-top:0.5rem;line-height:1.5">' + data.desc + '</p>';
        tooltip.style.left = rect.left + 'px';
        tooltip.style.top = (rect.bottom + 8) + 'px';
        tooltip.style.opacity = '1';
        setTimeout(function() { tooltip.style.opacity = '0'; }, 4000);
      }
    });
  });

  /* ===== SMOOTH SCROLL ===== */
  document.querySelectorAll('a[href^="#"]').forEach(function(link) {
    link.addEventListener('click', function(e) {
      e.preventDefault();
      var targetId = this.getAttribute('href').substring(1);
      var target = document.getElementById(targetId);
      if (target) {
        var top = target.getBoundingClientRect().top + window.pageYOffset - 80;
        window.scrollTo({ top: top, behavior: 'smooth' });
      }
    });
  });

  /* ===== FAQ ACCORDION (works with existing data-state attributes) ===== */
  var faqSection = document.getElementById('faq');
  if (faqSection) {
    var faqButtons = faqSection.querySelectorAll('button[data-state]');
    faqButtons.forEach(function(btn) {
      btn.addEventListener('click', function() {
        var h3 = this.closest('h3');
        if (!h3) return;
        var contentDiv = h3.nextElementSibling;
        if (!contentDiv) return;

        var isOpen = this.getAttribute('data-state') === 'open';
        var icon = this.querySelector('svg');
        var parentItem = h3.parentElement;

        if (isOpen) {
          // Close
          contentDiv.removeAttribute('hidden');
          contentDiv.style.maxHeight = '0';
          contentDiv.style.overflow = 'hidden';
          contentDiv.style.opacity = '0';
          contentDiv.style.transition = 'max-height 0.3s ease, opacity 0.3s ease';
          this.setAttribute('data-state', 'closed');
          this.setAttribute('aria-expanded', 'false');
          if (contentDiv.getAttribute('data-state') !== null) contentDiv.setAttribute('data-state', 'closed');
          if (parentItem && parentItem.getAttribute('data-state') !== null) parentItem.setAttribute('data-state', 'closed');
          if (icon) icon.style.transform = 'rotate(0deg)';
          setTimeout(function() { contentDiv.setAttribute('hidden', ''); }, 300);
        } else {
          // Close all others first
          faqButtons.forEach(function(otherBtn) {
            if (otherBtn !== btn && otherBtn.getAttribute('data-state') === 'open') {
              otherBtn.click();
            }
          });

          // Open this one
          contentDiv.removeAttribute('hidden');
          contentDiv.style.overflow = 'hidden';
          contentDiv.style.maxHeight = '0';
          contentDiv.style.opacity = '0';
          contentDiv.style.transition = 'max-height 0.3s ease, opacity 0.3s ease';

          // Force reflow
          contentDiv.offsetHeight;

          contentDiv.style.maxHeight = contentDiv.scrollHeight + 'px';
          contentDiv.style.opacity = '1';
          this.setAttribute('data-state', 'open');
          this.setAttribute('aria-expanded', 'true');
          if (contentDiv.getAttribute('data-state') !== null) contentDiv.setAttribute('data-state', 'open');
          if (parentItem && parentItem.getAttribute('data-state') !== null) parentItem.setAttribute('data-state', 'open');
          if (icon) icon.style.transform = 'rotate(180deg)';
        }
      });
    });
  }

  /* ===== REVIEW CAROUSEL ===== */
  var carouselEl = document.querySelector('[data-carousel]');
  if (carouselEl) {
    var slides = carouselEl.querySelectorAll('[data-carousel-slide]');
    var allDots = document.querySelectorAll('[data-carousel-dot]');
    var counters = carouselEl.querySelectorAll('[data-carousel-counter]');
    var current = 0;
    var total = slides.length;

    function goToSlide(index) {
      if (index < 0) index = total - 1;
      if (index >= total) index = 0;
      current = index;

      slides.forEach(function(slide, i) {
        slide.style.display = i === current ? 'block' : 'none';
      });

      allDots.forEach(function(dot, i) {
        dot.className = dot.className.replace(/w-6|w-1\.5/g, '').replace(/bg-primary\/60|bg-\[hsl\(160,20%,75%\)\]/g, '').trim();
        if (i === current) {
          dot.style.width = '1.5rem';
          dot.style.backgroundColor = 'hsl(160, 35%, 35%)';
        } else {
          dot.style.width = '0.375rem';
          dot.style.backgroundColor = 'hsl(160, 20%, 75%)';
        }
      });

      counters.forEach(function(c) { c.textContent = (current + 1) + '/' + total; });
    }

    // Event delegation for prev/next buttons (they're inside each slide)
    carouselEl.addEventListener('click', function(e) {
      var prevBtn = e.target.closest('[data-carousel-prev]');
      var nextBtn = e.target.closest('[data-carousel-next]');
      if (prevBtn) goToSlide(current - 1);
      if (nextBtn) goToSlide(current + 1);
    });

    allDots.forEach(function(dot, i) {
      dot.addEventListener('click', function() { goToSlide(i); });
    });

    goToSlide(0);
    setInterval(function() { goToSlide(current + 1); }, 8000);
  }

  /* ===== CTA BUTTONS → scroll to form ===== */
  document.querySelectorAll('button').forEach(function(btn) {
    var text = (btn.textContent || '').trim().toLowerCase();
    if (text === 'записаться' || text === 'записаться на интенсив' || text === 'получить программу') {
      btn.addEventListener('click', function() {
        var form = document.getElementById('cta-form');
        if (form) {
          var top = form.getBoundingClientRect().top + window.pageYOffset - 80;
          window.scrollTo({ top: top, behavior: 'smooth' });
        }
      });
    }
  });

  /* ===== NAVBAR SCROLL EFFECT ===== */
  var nav = document.querySelector('nav');
  if (nav) {
    window.addEventListener('scroll', function() {
      if (window.scrollY > 50) {
        nav.style.backgroundColor = 'hsla(160, 55%, 5%, 0.95)';
        nav.style.backdropFilter = 'blur(12px)';
        nav.style.boxShadow = '0 4px 12px rgba(0,0,0,0.15)';
      } else {
        nav.style.backgroundColor = '';
        nav.style.backdropFilter = '';
        nav.style.boxShadow = '';
      }
    });
  }

  /* ===== FIX PROGRESS BARS (student stats) ===== */
  // Find all gradient bars (inner bars of progress indicators)
  var allBars = document.querySelectorAll('[class*="bg-gradient-to-r"][class*="from-primary"]');
  allBars.forEach(function(bar) {
    // Find the percentage text in the same row
    var row = bar.closest('[class*="flex"][class*="items-center"]');
    if (!row) row = bar.parentElement?.parentElement;
    if (!row) return;
    var texts = row.querySelectorAll('*');
    for (var i = 0; i < texts.length; i++) {
      var t = texts[i].textContent.trim();
      var match = t.match(/^(\d+)%$/);
      if (match) {
        var pct = parseInt(match[1]);
        bar.style.setProperty('width', pct + '%', 'important');
        bar.style.setProperty('min-width', pct > 0 ? '8px' : '0', 'important');
        bar.style.setProperty('transition', 'width 1s ease-out');
        break;
      }
    }
  });

});
